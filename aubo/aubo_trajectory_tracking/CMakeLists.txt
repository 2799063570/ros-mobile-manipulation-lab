cmake_minimum_required(VERSION 3.5)
project(aubo_trajectory_tracking)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs std_msgs std_srvs nav_msgs visualization_msgs kdl_parser aubo_ros_control actionlib moveit_core moveit_ros_planning)
find_package(orocos_kdl REQUIRED)
find_package(Eigen3 REQUIRED)
catkin_package()
add_executable(trajectory_tracker_node src/trajectory_tracker.cpp)
target_compile_features(trajectory_tracker_node PRIVATE cxx_std_14)
target_include_directories(trajectory_tracker_node PRIVATE ${catkin_INCLUDE_DIRS} ${orocos_kdl_INCLUDE_DIRS})
target_link_libraries(trajectory_tracker_node ${catkin_LIBRARIES} ${orocos_kdl_LIBRARIES})
add_executable(trajectory_preparation_node src/trajectory_preparation.cpp)
add_dependencies(trajectory_preparation_node ${catkin_EXPORTED_TARGETS})
target_compile_features(trajectory_preparation_node PRIVATE cxx_std_14)
target_include_directories(trajectory_preparation_node PRIVATE ${catkin_INCLUDE_DIRS} ${orocos_kdl_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
target_link_libraries(trajectory_preparation_node ${catkin_LIBRARIES} ${orocos_kdl_LIBRARIES})
catkin_install_python(PROGRAMS scripts/tracking_panel.py scripts/tracking_plot.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS trajectory_tracker_node trajectory_preparation_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  catkin_add_gtest(tracking_control_test test/tracking_control_test.cpp)
  if(TARGET tracking_control_test)
    target_compile_features(tracking_control_test PRIVATE cxx_std_14)
    target_include_directories(tracking_control_test PRIVATE src ${catkin_INCLUDE_DIRS} ${orocos_kdl_INCLUDE_DIRS})
    target_link_libraries(tracking_control_test ${catkin_LIBRARIES} ${orocos_kdl_LIBRARIES})
  endif()
  catkin_install_python(PROGRAMS test/tracking_test.py test/preparation_test.py test/tracking_clock_test.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  add_rostest(test/tracking_clock.test DEPENDENCIES trajectory_tracker_node)
  add_rostest(test/tracking.test DEPENDENCIES trajectory_tracker_node)
  add_rostest(test/preparation.test DEPENDENCIES trajectory_tracker_node trajectory_preparation_node aubo_cartesian_jog_node)
endif()
